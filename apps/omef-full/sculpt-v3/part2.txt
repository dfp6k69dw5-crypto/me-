const detailGain=v=>v<=0?0:Math.pow(clamp(v,0,1),.72);

drawParticles=function(map){
  const g=detailGain(params.particles);if(!g)return;ctx.save();ctx.globalCompositeOperation='screen';
  const target=160+Math.round(760*g),stride=Math.max(4,Math.floor(records.length/target)),cloud=2+Math.round(6*g);
  for(let i=0;i<records.length;i+=stride){let r=records[i],c=map(r.x,r.y),unc=1-r.D[0],spread=(2+22*g)*(.45+1.25*unc),ang=(i*.618+seed)*2.1,co=Math.cos(ang),si=Math.sin(ang);
    for(let k=0;k<cloud;k++){let rr=spread*(.18+.14*k),a=ang+k*2.399+i*.0017,dx=Math.cos(a)*rr*(1+.8*(1-r.D[2])),dy=Math.sin(a)*rr*(.45+.3*g),alpha=.018+.12*g*(.45+.55*unc);ctx.fillStyle='rgba(205,220,231,'+alpha+')';ctx.beginPath();ctx.arc(c[0]+dx*co-dy*si,c[1]+dx*si+dy*co,.45+1.15*g+(k===0?.35:0),0,TAU);ctx.fill();}
  }ctx.restore();
};

drawBraids=function(map){
  const g=detailGain(params.braids);if(!g)return;ctx.save();ctx.globalCompositeOperation='screen';
  const count=1+Math.round(4*g),target=900+Math.round(1600*g),stride=Math.max(1,Math.floor(records.length/target)),paths=Array.from({length:count},()=>[]);
  for(let i=2;i<records.length-2;i+=stride){let r=records[i],p=map(r.x,r.y),pa=map(records[i-2].x,records[i-2].y),pb=map(records[i+2].x,records[i+2].y),dx=pb[0]-pa[0],dy=pb[1]-pa[1],ll=Math.hypot(dx,dy)||1,nx=-dy/ll,ny=dx/ll;
    for(let b=0;b<count;b++){let zi=(b*7)%36,z=r.Z[zi]+.7*r.Z[(zi+2)%36],wave=Math.sin(i*.017+b*1.7+seed*.13)*4*g,off=z*(3+24*g)+wave;paths[b].push([p[0]+nx*off,p[1]+ny*off]);}}
  for(let b=0;b<count;b++){ctx.strokeStyle='rgba('+(218+b*5)+','+(224+b*4)+','+(234-b*2)+','+(.04+.20*g)+')';ctx.lineWidth=.55+1.45*g+.08*b;ctx.beginPath();paths[b].forEach((p,i)=>i?ctx.lineTo(p[0],p[1]):ctx.moveTo(p[0],p[1]));ctx.stroke();}
  ctx.restore();
};
