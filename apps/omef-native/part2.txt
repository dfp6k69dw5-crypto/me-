  return {seed, step:0, regime:clamp(regime|0,0,3), U, Ustar:U.map((x,i)=>x+.025*Math.sin(i+1)), L:Array(N).fill(0), theta, s, sigma: s.map(Math.exp), zS,zL, history, hHistory:Array(13).fill(U[9]), F:[0,0,0], obsPrev:[0,0,0,0,0], microMoments:Array.from({length:N},()=>[0,1,0,3]), tauCal:1, nuCal:12, coverage:.8, tailRisk:.08, releaseSurprise:0, revisionDiag:Array(N).fill(0), blocks:Array(N).fill(0), blockCount:1, benchmarkAdvantage:0};
}

function computeMarketWeights(U, market, deltaSpec, cRisk, theta, p){
  const scores=Array.from({length:N},()=>Array(N).fill(-99));
  for(let i=0;i<N;i++)for(let j=0;j<N;j++)if(i!==j){
    const asym=(i-j)/N;
    const base=.34*Math.sin((i+1)*(j+2)*.37)+.20*theta[i][j];
    const diff=Math.abs(U[i]-U[j]);
    const mr=.23*market[(i+j)%market.length] + .14*market[(i*3+j)%market.length];
    scores[i][j]=p.marketCoupling*(base+mr+.44*diff+.32*cRisk+.25*Math.max(-deltaSpec,0)+.16*asym);
  }
  return normalizeRowsZeroDiag(scores);
}

function laplacianDiagnostics(W, harmonicDiffusion){
  const S=Array.from({length:N},()=>Array(N).fill(0));
  for(let i=0;i<N;i++)for(let j=0;j<N;j++)S[i][j]=.5*(W[i][j]+W[j][i]);
  const L=Array.from({length:N},()=>Array(N).fill(0));
  for(let i=0;i<N;i++){
    let d=0; for(let j=0;j<N;j++)d+=S[i][j];
    for(let j=0;j<N;j++)L[i][j]=(i===j?d:0)-S[i][j];
  }
  const ev=jacobiEigenSymmetric(L,80,false).values;
  const lambda2=Math.max(0,ev[1]||0);
  const kcon=Math.sqrt(lambda2/(Math.max(.08,harmonicDiffusion)+1e-6));
  const cRisk=Math.max(.42-kcon,0);
  return {lambda2,kcon,cRisk,L};
}

function trajectoryDiagnostics(history, rankGuard){
  const K=Math.min(13,history.length), slice=history.slice(-K);
  const mu=Array(N).fill(0); for(const u of slice)for(let i=0;i<N;i++)mu[i]+=u[i]/K;
  const C=Array.from({length:N},()=>Array(N).fill(0));
  for(const u of slice)for(let i=0;i<N;i++)for(let j=0;j<N;j++)C[i][j]+=(u[i]-mu[i])*(u[j]-mu[j])/K;
  const vals=jacobiEigenSymmetric(C,80,false).values.map(x=>Math.max(0,x));
  const svals=vals.map(Math.sqrt).sort((a,b)=>b-a), top=svals[0]||1e-9, threshold=.055*rankGuard;
  const eff=svals.reduce((n,x)=>n+(x/(top+1e-9)>threshold?1:0),0);
  const psi=svals[svals.length-1]||0;
  return {psi,eff,svals};
}

function selectBlocks(U,W,L,p){
  // Implements the source idea A^B = weighted posterior/market/harmonic affinity and chooses G by cut+complexity+balance.
  const A=Array.from({length:N},()=>Array(N).fill(0));
  const sd=Math.sqrt(variance(U))+1e-6;
  for(let i=0;i<N;i++)for(let j=0;j<N;j++)if(i!==j){
    const corr=Math.exp(-Math.abs(U[i]-U[j])/(1.1*sd));
    const market=.5*(W[i][j]+W[j][i]);
    const harmonic=Math.exp(-Math.abs(L[i][j])*2.2);
    A[i][j]=.36*corr+.42*market+.22*harmonic;
  }
  // Agglomerative approximation to SpecClust_G: deterministic, tiny N, scored with cut + complexity + balance.
  const maxG=5; let best=null;
  for(let G=1;G<=maxG;G++){
    let clusters=Array.from({length:N},(_,i)=>[i]);
    while(clusters.length>G){
      let bi=0,bj=1,bs=-Infinity;
      for(let i=0;i<clusters.length;i++)for(let j=i+1;j<clusters.length;j++){
        let s=0,c=0; for(const a of clusters[i])for(const b of clusters[j]){s+=A[a][b];c++;}
        s/=c||1; if(s>bs){bs=s;bi=i;bj=j;}
      }
      clusters[bi]=clusters[bi].concat(clusters[bj]); clusters.splice(bj,1);
    }
    let cut=0; for(let i=0;i<N;i++)for(let j=i+1;j<N;j++){
      const ci=clusters.findIndex(c=>c.includes(i)), cj=clusters.findIndex(c=>c.includes(j)); if(ci!==cj)cut+=A[i][j];
    }
    const balance=clusters.reduce((s,c)=>s+Math.pow(c.length-N/G,2),0);
    const complexity=.28*p.blockGranularity*G;
    const score=cut + complexity + .08*p.blockGranularity*balance;
    if(!best||score<best.score)best={score,G,clusters};
  }
  const ids=Array(N).fill(0); best.clusters.forEach((c,g)=>c.forEach(i=>ids[i]=g));
  return {ids,count:best.G,affinity:A};
}

function particleDiagnostics(state,U,p,rnd,blocks){
  // Full 5,000-particle synthetic state cloud for the visual model: each particle carries all ten organ coordinates.
  // We use the OMEF weight -> normalize -> ESS -> weighted moments structure, while the observation target is closed-loop/synthetic.
  const vals=new Float32Array(P*N), weights=new Float64Array(P); let sum=0,sum2=0;
  const sharp=1.7/Math.max(.12,p.particleDiversity), sigmaBase=Math.max(.035,mean(state.sigma)*p.volatility);
  for(let q=0;q<P;q++){
    let mahal=0;
    for(let i=0;i<N;i++){
      const sig=sigmaBase*(.72+.10*i/N)*(1+.25*(state.revisionDiag?.[i]||0));
      const z=(rnd()+rnd()+rnd()+rnd()-2)*sig;
      const x=U[i]+z+.035*Math.sin((q+1)*(i+2)*.013); vals[q*N+i]=x;
      const resid=x-U[i]+.10*state.releaseSurprise*p.revisionRisk*(i===4||i===7?1:.35); mahal+=resid*resid/(sig*sig+.025);
    }
    const w=Math.exp(-.5*sharp*mahal/N);weights[q]=w;sum+=w;sum2+=w*w;
  }
  const neff=sum*sum/(sum2+1e-12), dpf=clamp(1-neff/P,0,1);
  const moments=Array.from({length:N},()=>[0,0,0,0]);
  for(let q=0;q<P;q++){
    const w=weights[q]/(sum+1e-12);
    for(let i=0;i<N;i++){const x=vals[q*N+i]; moments[i][0]+=w*x; moments[i][1]+=w*x*x; moments[i][2]+=w*x*x*x; moments[i][3]+=w*x*x*x*x;}
  }
  for(let i=0;i<N;i++){
    const m=moments[i][0],v=Math.max(1e-8,moments[i][1]-m*m),sd=Math.sqrt(v);
    const mu3=moments[i][2]-3*m*moments[i][1]+2*m*m*m;
    const mu4=moments[i][3]-4*m*moments[i][2]+6*m*m*moments[i][1]-3*m*m*m*m;
    moments[i]=[m,sd,mu3/(sd*sd*sd+1e-9),mu4/(v*v+1e-9)];
  }
  const blockDpf=[];
  if(blocks){
    const G=1+Math.max(...blocks); for(let g=0;g<G;g++){
      const ids=[];for(let i=0;i<N;i++)if(blocks[i]===g)ids.push(i);let sw=0,sw2=0;
      for(let q=0;q<P;q++){
        let e=0;for(const i of ids){const dx=vals[q*N+i]-U[i];e+=dx*dx/(state.sigma[i]*state.sigma[i]+.025);}const w=weights[q]*Math.exp(-.12*e/(ids.length||1));sw+=w;sw2+=w*w;
      }
      const ne=sw*sw/(sw2+1e-12);blockDpf.push(clamp(1-ne/P,0,1));
    }
  }
  return {neff,dpf,moments,blockDpf};
}
