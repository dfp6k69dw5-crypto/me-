function selfTest(){
  const failures=[];
  // Invariants over multiple seeds/regimes.

    if(!s.U.every(Number.isFinite))failures.push(`non-finite U seed ${seed} regime ${regime}`);
    for(let i=0;i<N;i++){const rs=s.frame.W[i].reduce((a,b)=>a+b,0); if(Math.abs(rs-1)>1e-8)failures.push(`W row sum ${i}`); if(Math.abs(s.frame.W[i][i])>1e-12)failures.push(`W diagonal ${i}`);}

    if(!(s.frame.confidence>0&&s.frame.confidence<=1.000001))failures.push('confidence out of range');

  }

  return {pass:failures.length===0, failures, audit, visibleControls:visible.map(a=>a.id), sourceBlocks:SOURCE_MAP.length};
}
