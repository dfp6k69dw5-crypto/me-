    shape:deriveShape(s.U,W,cyc,s.blocks,{...p}, {humility,confidence,dpf:pd.dpf,deltaSpec,cRisk:lap.cRisk,rankRisk:finalRankRisk,releaseSurprise:s.releaseSurprise,sigmaMean:mean(s.sigma),blockCount:s.blockCount,microSkew,microKurt,factorEnergy,observability:os.observability,spillover:os.spillover,benchmarkLoss:Math.max(-obs.advantage,0)/(1+Math.abs(obs.advantage)),complexity:complexity.normalized,regimeEntropy})
  };
  return s;
}

function deriveShape(U,W,cyc,blocks,p,d){
  let asym=0; for(let i=0;i<N;i++)for(let j=i+1;j<N;j++)asym+=Math.abs(W[i][j]-W[j][i]); asym/=45;
  const cEnergy=Math.sqrt(mean(cyc.slice(0,9).map(x=>x*x)));
  const uSpread=Math.sqrt(variance(U));
  return {
    coupling:clamp(.18+.32*p.marketCoupling+.42*asym,0,1.8),
    interaction:clamp(.18+.46*p.interaction+.12*uSpread,0,2),
    cycle:clamp(.18+.46*cEnergy,0,2.3),
    detune:p.cycleDetune,
    memory:clamp(.12+.52*p.fibonacciMemory+.18*Math.abs(U[9]),0,2.2),
    hidden:clamp(.12+.58*p.hiddenStrain+.22*Math.abs(U[9]),0,2.4),
    spread:clamp(.08+.42*p.volatility*d.sigmaMean+.28*p.revisionRisk*Math.abs(d.releaseSurprise),.04,1.8),
    tear:clamp(.04+.40*p.shockEnergy*(1-d.confidence)+.60*d.dpf+.24*p.revisionRisk,0,2.0),
    brake:clamp(.12+.55*p.stabilityBrake*Math.max(-d.deltaSpec,0)+.35*p.stabilityBrake*d.rankRisk,0,2),
    connectivity:clamp(.25+.60*p.harmonicDiffusion*(.25+d.cRisk),0,2.2),
    humility:clamp(.08+p.humilityPressure*d.humility,0,2.5),
    blocks:clamp(.18+.20*d.blockCount*p.blockGranularity,0,1.8),
    accounting:clamp(.1+.38*p.accountingConstraint,0,1.6),
    rank:clamp(.08+.65*p.rankGuard*d.rankRisk,0,2),
    micro:clamp(.08+.34*d.microSkew+.20*d.microKurt,0,1.8),
    factor:clamp(.08+.34*d.factorEnergy,0,1.8),
    spillover:clamp(.10+.72*d.spillover,0,1.5),
    observability:clamp(.25+.75*d.observability,0,1.2),
    benchmark:clamp(.06+.85*d.benchmarkLoss,0,1.6),
    complexity:clamp(.10+.78*d.complexity,0,1.5),
    regimeEntropy:clamp(d.regimeEntropy,0,1),
    regime:U.length?0:0
  };
}

const PROJ = Array.from({length:N},(_,i)=>{
  const a=(i+.5)*2.399963229728653, z=1-2*(i+.5)/N, r=Math.sqrt(Math.max(0,1-z*z));
  return [r*Math.cos(a),r*Math.sin(a),z];
});

function samplePoint(frame,id,time=0){
  const h1=hash01(id,11), h2=hash01(id,29), h3=hash01(id,47), h4=hash01(id,83);
  const t=TAU*h1, q=(h2*2-1), r=Math.sqrt(h3), branch=Math.floor(h4*Math.max(1,frame.blockCount));
  const d=Array(N), coupled=Array(N);
  for(let i=0;i<N;i++){
    const block=frame.blocks[i]||0;
    const phase=.31*i + .37*frame.cycles[i] + .18*block + time*(.03+.004*i);
    const harmonic=Math.sin((1+.17*i)*t+phase)+.42*Math.cos((2+.11*i)*t*(1+.08*frame.shape.detune)+q*(i+1));
    const strand=(.28+.10*frame.shape.micro)*q*Math.sin((i+2)*t+branch*.9)+.08*frame.shape.factor*Math.sin(t*(1.3+.11*i)+q*2.0);
    d[i]=harmonic+strand+.56*frame.U[i]+.24*frame.cycles[i];
  }
  for(let i=0;i<N;i++){
    let a=0; for(let j=0;j<N;j++)a+=frame.W[i][j]*Math.tanh(d[j]);
    coupled[i]=mix(d[i],a,clamp(frame.shape.coupling*.42,0,.92));
  }
  const pre=coupled.slice();
  for(let i=0;i<N;i++) coupled[i]=Math.tanh(pre[i]+.20*frame.shape.interaction*Math.sin(pre[i]*pre[(i+3)%N]+t));
  let x=0,y=0,z=0;
  for(let i=0;i<N;i++){
    const v=coupled[i]*(.22+.10*frame.shape.observability+.14*Math.abs(frame.U[i])) + .08*q*(frame.blocks[i]===branch?1:-.25);
    x+=PROJ[i][0]*v; y+=PROJ[i][1]*v; z+=PROJ[i][2]*v;
  }
  const sh=frame.shape;
  const ang=.36*sh.cycle*Math.sin(t*(1.0+.35*sh.detune))+.26*sh.hidden*Math.sin(z*2.2+t*.5)+.20*frame.regime;
  const ca=Math.cos(ang),sa=Math.sin(ang); let X=ca*x-sa*y, Y=sa*x+ca*y;
  X += .32*sh.blocks*Math.sin((frame.blockCount+1+sh.complexity)*t+branch) + .22*sh.connectivity*Math.sin(Y*2.1+t)+.16*sh.spillover*Math.sin(z*2.7-t);
  Y += .28*sh.accounting*Math.sin(X*1.7-t*.7) - .14*sh.rank*Math.sin(3*t);
  z += .34*sh.memory*Math.sin(t*PHI + frame.hidden) + .20*sh.interaction*Math.sin(X*Y*2.3)+.16*sh.factor*Math.cos(t*(PHI+.2*sh.regimeEntropy));
  const pinch=1/(1+.22*sh.humility*Math.abs(z)); X*=pinch;Y*=pinch;
  const spread=.08+sh.spread*.22; X+=spread*q*Math.cos(t*3.1);Y+=spread*q*Math.sin(t*2.7);z+=spread*(r-.5)*Math.sin(t*1.9);
  const tear=(sh.tear+.34*sh.benchmark)*.22*(h4>.76?1:0); X+=tear*Math.sign(Math.sin(t*frame.blockCount+1))*q; z+=tear*Math.sin(7*t+branch);
  const brake=1/(1+.20*sh.brake*(X*X+Y*Y+z*z));
  return [X*brake,Y*brake,z*brake];
}

function makeFrame(params={},seed=12345,regime=0,steps=9){
  let s=createState(seed,regime); for(let i=0;i<steps;i++)s=stepState(s,params); return s.frame;
}

function visualSignature(params={},seed=12345,regime=0,count=448){
  const f=makeFrame(params,seed,regime,8); let sx=0,sy=0,sz=0,sxx=0,syy=0,szz=0,sxy=0; const pts=[];
  for(let i=0;i<count;i++){const p=samplePoint(f,i,0);pts.push(p);sx+=p[0];sy+=p[1];sz+=p[2];sxx+=p[0]*p[0];syy+=p[1]*p[1];szz+=p[2]*p[2];sxy+=p[0]*p[1];}
  const n=count,mx=sx/n,my=sy/n,mz=sz/n;
  return {frame:f, pts, stats:[mx,my,mz,sxx/n-mx*mx,syy/n-my*my,szz/n-mz*mz,sxy/n-mx*my, f.blockCount, f.effectiveRank, f.dpf, f.confidence]};
}

function signatureDistance(a,b){
  const n=Math.min(a.pts.length,b.pts.length); let rms=0,rad=0;
  for(let i=0;i<n;i++){const A=a.pts[i],B=b.pts[i];const dx=A[0]-B[0],dy=A[1]-B[1],dz=A[2]-B[2];rms+=dx*dx+dy*dy+dz*dz;rad+=A[0]*A[0]+A[1]*A[1]+A[2]*A[2];}
  rms=Math.sqrt(rms/n); rad=Math.sqrt(rad/n)+1e-6;
  let stat=0; for(let i=0;i<a.stats.length;i++)stat+=Math.abs(a.stats[i]-b.stats[i])/(.15+Math.abs(a.stats[i])); stat/=a.stats.length;
  return rms/rad + .22*stat;
}

function auditControls(seed=12345){
  const base={...DEFAULT_PARAMS}, out=[];
  for(const d of CONTROL_DEFS){
    const lo={...base,[d.id]:d.min}, hi={...base,[d.id]:d.max};
    const A=visualSignature(lo,seed,1,320), B=visualSignature(hi,seed,1,320); const score=signatureDistance(A,B);
    out.push({...d,impact:score,pass:Number.isFinite(score)&&score>=.115});
  }
  return out;
}

function quality(frame,count=768){
  let minX=Infinity,maxX=-Infinity,minY=Infinity,maxY=-Infinity,s=0,s2=0,finite=true;
  const bins=new Uint16Array(64);
  for(let i=0;i<count;i++){
    const p=samplePoint(frame,i); if(!p.every(Number.isFinite)){finite=false;break;}
    minX=Math.min(minX,p[0]);maxX=Math.max(maxX,p[0]);minY=Math.min(minY,p[1]);maxY=Math.max(maxY,p[1]); const rr=p[0]*p[0]+p[1]*p[1]+p[2]*p[2];s+=rr;s2+=rr*rr;
    const bx=clamp(Math.floor((p[0]+3)/6*8),0,7),by=clamp(Math.floor((p[1]+3)/6*8),0,7);bins[by*8+bx]++;
  }
