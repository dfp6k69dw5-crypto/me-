/* a constant term's weight is inert: always for a constant of 0,
   and under mul for a constant of 1, since 1−w+w = 1                */
const weightless = (T, t, op) => !!T[t].noW || (!!T[t].solo && op === "mul");
function blend(op, vals, ws) {
    if (op === "mul") {
        let acc = 1, any = false;
        for (let i = 0; i < vals.length; i++) {
            if (vals[i] === null)
                continue;
            any = true;
            acc *= 1 - ws[i] + ws[i] * vals[i];
        }
        return any ? acc : 1;
    }
    if (op === "max") {
        let acc = -1e9, any = false;
        for (let i = 0; i < vals.length; i++) {
            if (vals[i] === null)
                continue;
            any = true;
            acc = Math.max(acc, ws[i] * vals[i]);
        }
        return any ? acc : 0;
    }
    let acc = 0;
    for (let i = 0; i < vals.length; i++)
        if (vals[i] !== null)
            acc += ws[i] * vals[i];
    return acc;
}
const evalW = (u, m) => Math.max(0, blend(m.op, m.slots.map((s) => (s.t === "off" ? null : W_T[s.t].f(u, s.p))), m.slots.map((s) => s.w)));
const evalK = (s, m, ctx) => blend(m.op, m.slots.map((sl) => (sl.t === "off" ? null : K_T[sl.t].f(s, sl.p, ctx))), m.slots.map((sl) => sl.w)) * 3;
const evalZ = (ctx, m) => blend(m.op, m.slots.map((s) => (s.t === "off" ? null : Z_T[s.t].f(ctx, s.p))), m.slots.map((s) => s.w));
function warp(x, y, m) {
    let ax = 0, ay = 0, any = false;
    for (let i = 0; i < m.slots.length; i++) {
        const sl = m.slots[i];
        if (sl.t === "off")
            continue;
        any = true;
        const o = V_T[sl.t].f(x, y, sl.p);
        if (!isFinite(o[0]) || !isFinite(o[1]))
            continue;
        ax += sl.w * o[0];
        ay += sl.w * o[1];
    }
    if (!any)
        return [x, y];
    const r = Math.hypot(ax, ay);
    if (r > 8) {
        ax = (ax / r) * 8;
        ay = (ay / r) * 8;
    }
    return [ax, ay];
}

function stemCurve(P, lenScale, ll) {
    const L = P.length * (lenScale || 1);
    const steps = Math.round(160 + L * 34);
    const h = L / steps;
    const acc = (kk) => P.lambda * kk - 0.5 * kk * kk * kk;
    let k = P.k0, kp = P.kp0, th = 0, x = 0, y = 0, s = 0;
    const pts = [[0, 0, 0]];
    for (let i = 0; i < steps; i++) {
        const a1 = kp, b1 = acc(k);
        const a2 = kp + (h / 2) * b1, b2 = acc(k + (h / 2) * a1);
        const a3 = kp + (h / 2) * b2, b3 = acc(k + (h / 2) * a2);
        const a4 = kp + h * b3, b4 = acc(k + h * a3);
        k += (h / 6) * (a1 + 2 * a2 + 2 * a3 + a4);
        kp += (h / 6) * (b1 + 2 * b2 + 2 * b3 + b4);
        if (!isFinite(k) || Math.abs(k) > 40)
            break;
        s += h;
        th += evalK(s, P.mixK, { k, L }) * h;
        x += Math.cos(th) * h;
        y += Math.sin(th) * h;
        pts.push([x, y, evalZ({ u: s / L, r: cl01(Math.hypot(x, y) / Math.max(0.001, L)), theta: th, ll }, P.mixZ)]);
    }
    return pts;
}
function petalParts(P, over, ll) {
    const Q = over ? Object.assign({}, P, over) : P;
    const res = 92;
    const g = (u) => Math.exp(-Math.pow((1 - u) / 0.22, 2));
    const X = (u) => u - Q.notch * g(u);
    const fy = Math.cos(Q.fold);
    const sd = Q.seed || 0;
    const pt = (u, sgn) => {
        const wob = 1 + Q.margin * 0.3 * snoise(u * 6.2 + (sgn > 0 ? 0 : 3.7), sd);
        const y = sgn * evalW(u, Q.mixW) * Q.pW * fy * wob;
        const x = X(u) + Q.shear * y;
        return [x, y, evalZ({ u, r: cl01(Math.hypot(x, y)), theta: Math.atan2(y, x), ll }, Q.mixZ)];
    };
    const outline = [];
    for (let i = 0; i <= res; i++)
        outline.push(pt(i / res, 1));
    for (let i = res; i >= 0; i--)
        outline.push(pt(i / res, -1));
    const parts = [outline];
    const mid = [];
    for (let i = 0; i <= 26; i++) {
        const u = i / 26;
        mid.push([X(u), 0, evalZ({ u, r: u, theta: 0, ll }, Q.mixZ)]);
    }
    parts.push(mid);
    for (let k = 0; k < Q.pleats; k++) {
        const u = 0.13 + ((k + 1) / (Q.pleats + 1)) * 0.36;
        parts.push([[0, 0, 0], pt(u, 1)]);
        parts.push([[0, 0, 0], pt(u, -1)]);
    }
    return parts;
}
function veinParts(Q) {
    const out = [];
    const n = Math.round(Q.veinN);
    if (n < 1)
        return out;
    const g = (u) => Math.exp(-Math.pow((1 - u) / 0.22, 2));
    const X = (u) => u - Q.notch * g(u);
    const fy = Math.cos(Q.fold);
    const sd = Q.seed || 0;
    const wAt = (u) => evalW(u, Q.mixW) * Q.pW * fy * (1 + Q.margin * 0.3 * snoise(u * 6.2, sd));
    for (let k = 0; k < n; k++) {
        const u0 = 0.07 + ((k + 0.5) / n) * 0.74 + (hash(sd, k) - 0.5) * 0.05;
        const uEnd = Math.min(0.97, u0 + Q.veinReach * (1 - u0));
        for (let si = 0; si < 2; si++) {
            const sgn = si ? -1 : 1;
            const pts = [];
            for (let i = 0; i <= 16; i++) {
                const t = i / 16;
                const u = u0 + (uEnd - u0) * t;
                const y = sgn * wAt(u) * (0.94 * t) * (0.88 + 0.12 * Math.sin(t * 2.6));
                pts.push([X(u) + Q.shear * y, y, 0]);
            }
            out.push(pts);
            if (Q.veinBranch > 0.5 && k % 2 === 0) {
                const ub = u0 + (uEnd - u0) * 0.5, bs = [];
                for (let i = 0; i <= 9; i++) {
                    const t = i / 9;
                    const u = ub + (0.985 - ub) * t * 0.8;
                    const y = sgn * wAt(u) * (0.42 + 0.5 * t);
                    bs.push([X(u) + Q.shear * y, y, 0]);
                }
                out.push(bs);
            }
        }
    }
    return out;
}
function keelParts(P, ll) {
    const res = 60, up = [], lo = [];
    for (let i = 0; i <= res; i++) {
        const u = i / res, drift = P.keelRise * u * u;
        const wv = evalW(u, P.mixW) * P.pW;
        const zz = evalZ({ u, r: u, theta: 0, ll }, P.mixZ);
        up.push([u, drift - wv * 1.15, zz]);
        lo.push([u, drift + wv * 0.45, zz]);
    }
    return [up.concat(lo.reverse())];
}
function place(parts, rot, scale, tx, ty, zAdd) {
    return parts.map((pl) => pl.map((p) => {
        const q = rot2(p[0] * scale, p[1] * scale, rot);
        return [q[0] + tx, q[1] + ty, p[2] * scale + zAdd];
    }));
}
/* ── the recursive type ─────────────────────────────────────────
   flower = centre ⫐ {petal} ,  petal = region that may hold a flower
   depth 1 is the old one-level generator, unchanged.              */
function petalSlots(P, level) {
    const UP = -Math.PI / 2;
    if (level > 0) {
        const N = Math.max(1, Math.round(P.childN));
        const out = [];
        for (let k = 0; k < N; k++)
            out.push({ rot: UP + (k * 2 * Math.PI) / N, s: 1, arm: k, d: 0.4 + level * 0.15, keel: false });
        return out;
    }
    if (P.arrangement === "papilionaceous") {
        const wa = (P.wingAngle * Math.PI) / 180;
        return [
            { rot: UP, s: 1, arm: 0, d: 0, keel: false },
            { rot: UP + wa, s: 0.78, arm: 1, d: 0.55, keel: false },
            { rot: UP - wa, s: 0.78, arm: 1, d: 0.58, keel: false },
            { rot: UP + (P.keelAngle * Math.PI) / 180, s: 0.6, arm: 2, d: 1, keel: true },
        ];
    }
    if (P.arrangement === "cruciform") {
        const out = [];
        for (let k = 0; k < 4; k++)
            out.push({ rot: UP + (k * Math.PI) / 2, s: 1, arm: k, d: (k / 3) * 0.85, keel: false });
        return out;
    }
    const N = P.symmetry, out = [];
    for (let k = 0; k < N; k++)
        out.push({ rot: UP + (k * 2 * Math.PI) / N, s: 1, arm: k, d: N === 1 ? 0.5 : k / (N - 1), keel: false });
    return out;
}
const CAP = 1400;
function emitFlower(P, level, sc, tx, ty, baseRot, ll, zAdd, out) {
    const slots = petalSlots(P, level);
    const detail = level === 0 || P.childDetail;
    for (let n = 0; n < slots.length; n++) {
        const sl = slots[n];
        if (out.length > CAP)
            return;
        const sd = level * 17.3 + n * 31.7 + 7.1;
        const g = P.irregular;
        const h = (i) => hash(sd, i);
        const over = {
            pleats: detail ? P.pleats : 0,
            seed: h(5) * 9,
            pW: P.pW * (1 + (h(1) - 0.5) * 0.45 * g),
            notch: cl01(P.notch * (1 + (h(2) - 0.5) * 0.9 * g)),
            fold: Math.max(0, P.fold * (1 + (h(3) - 0.5) * 0.7 * g)),
            shear: P.shear + (h(4) - 0.5) * 0.3 * g,
        };
        const rotJ = (h(6) - 0.5) * 0.34 * g;
        const lenJ = 1 + (h(7) - 0.5) * 0.3 * g;
        const tone = (h(8) - 0.5);
        const R0 = baseRot + sl.rot + rotJ, S0 = sc * sl.s * lenJ;
        const parts = sl.keel ? keelParts(P, ll) : petalParts(P, over, ll);
        const kept = detail ? parts : [parts[0]];
        place(kept, R0, S0, tx, ty, zAdd).forEach((pl, i) => out.push({ pts: pl, arm: sl.arm + level * 3, depth: sl.d, tone, seed: sd,
            kind: i === 0 ? "blade" : "detail" }));
        if (detail && !sl.keel && P.veinN >= 1) {
            place(veinParts(Object.assign({}, P, over)), R0, S0, tx, ty, zAdd).forEach((pl) => out.push({ pts: pl, arm: sl.arm + level * 3, depth: sl.d + 0.002, tone, seed: sd, kind: "vein" }));
        }
        if (level + 1 < Math.round(P.depth) && out.length < CAP && !sl.keel) {
            const au = cl01(P.childAt);
            const lx = (au - P.notch * Math.exp(-Math.pow((1 - au) / 0.22, 2))) * S0;
            const CP = Object.assign({}, P, {
                shear: P.parity ? -P.shear : P.shear,
                fold: P.fold * P.childFold,
            });
            emitFlower(CP, level + 1, S0 * P.childScale, tx + Math.cos(R0) * lx, ty + Math.sin(R0) * lx, R0 + Math.PI / 2 + (P.childTwist * Math.PI) / 180, ll, zAdd, out);
        }
    }
}
function buildLayer(P, ll, zAdd, rotAdd, sc) {
    const out = [];
    if (P.mode === "petal") {
        emitFlower(P, 0, sc, 0, 0, rotAdd, ll, zAdd, out);
        return out;
    }
    const N = P.symmetry, K = P.strands;
    for (let j = 0; j < N; j++) {
        const armA = (j / N) * Math.PI * 2 - Math.PI / 2 + rotAdd;
        for (let k = 0; k < K; k++) {
            const fan = K === 1 ? 0 : ((k / (K - 1) - 0.5) * P.spread * Math.PI) / 180;
            const twist = K === 1 ? (j * P.spread * Math.PI) / (180 * Math.max(1, N)) : 0;
            const off = fan + twist;
            const t = K === 1 ? 0 : k / (K - 1) - 0.5;
            const raw = stemCurve(Object.assign({}, P, { k0: P.k0 * (1 + t * 0.55) }), 1 + t * 0.3, ll);
            const pts = raw.map((p) => { const q = rot2(p[0] * sc, p[1] * sc, armA + off); return [q[0], q[1], p[2] * sc + zAdd]; });
            out.push({ pts, arm: j, depth: 0.3, kind: "stem" });
            if (raw.length < 4)
                continue;
            const a = raw[raw.length - 1], b = raw[raw.length - 2];
            const ta = Math.atan2(a[1] - b[1], a[0] - b[0]) + armA + off;
            const tip = pts[pts.length - 1];
            if (P.mode === "bloom") {
                emitFlower(P, 0, P.petal * 3 * sc, tip[0], tip[1], ta + Math.PI / 2, ll, tip[2], out);
            }
            if (P.anthers) {
                const Lb = P.antherLen * sc, ang = ta + Math.PI / 2.6;
                const d = [Math.cos(ang) * Lb, Math.sin(ang) * Lb];
                out.push({ pts: [[tip[0] - d[0], tip[1] - d[1], tip[2]], [tip[0] + d[0], tip[1] + d[1], tip[2]]], arm: j, depth: 0.9, kind: "detail" });
            }
        }
    }
    return out;
}
function buildStrokes(P) {
    const L = Math.max(1, Math.round(P.layers));
    let all = [];
    for (let l = 0; l < L; l++) {
        const ll = L === 1 ? 0 : l / (L - 1);
        const drift = 1 + ll * P.layerDrift;
        const d = ll * P.layerDrift;
        const Pl = Object.assign({}, P, {
            k0: P.k0 * drift,
            pW: P.pW * (1 - d * 0.35),
            notch: cl01(P.notch * (1 + d * 1.6)),
            fold: Math.max(0, P.fold + d * 0.5),
            shear: P.shear + d * 0.3,
            length: P.length * (1 + d * 0.25),
        });
        all = all.concat(buildLayer(Pl, ll, ll * P.layerGap, (ll * P.layerTwist * Math.PI) / 180, 1 - ll * P.layerShrink));
    }
    const active = P.mixV.
