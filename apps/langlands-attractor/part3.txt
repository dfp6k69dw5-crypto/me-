    const c=Math.cos(phi*P.fm), s=Math.sin(phi*P.fm);

    let E=0;

    let nw=Math.tanh((.38+.52*(1-P.nilpotent))*WW + P.fm*(X*WW-Y*Z)*.22 + P.gerbe*Math.cos(rank*Z+rPhase) + P.verdier*E*.48);

    }

    I=clamp(Math.pow(clamp(I,0,1),P.gamma),.01,1);
    // projection also uses dual variables but no arbitrary post-render rotation.
    const theta=P.fm*(w*.72+z*.28)+P.gerbe*zPhase*.6;
    const cc=Math.cos(theta),ss=Math.sin(theta);
    const qx=cc*x-ss*y + .34*P.functorMix*z + .12*E + .16*P.zx*sat(z*x);
    const qy=ss*x+cc*y + .28*P.verdier*w - .1*E + .14*P.xy*sat(x*y);
    return [qx,qy,I,x,y,z,w];
  }
}
function sampleBounds(){

  }
  if(!pts.length||!(maxX>minX)||!(maxY>minY))return {cx:0,cy:0,s:Math.min(W,H)*.18};

}
