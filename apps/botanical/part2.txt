slots.some((s) => s.t !== "off");
    if (active) {
        let R = 0;
        for (const s of all)
            for (const p of s.pts)
                R = Math.max(R, Math.hypot(p[0], p[1]));
        R = Math.max(1e-4, R);
        for (const s of all)
            for (const p of s.pts) {
                const o = warp(p[0] / R, p[1] / R, P.mixV);
                p[0] = o[0] * R;
                p[1] = o[1] * R;
            }
    }
    for (const s of all) {
        let z = 0;
        for (const p of s.pts)
            z += p[2];
        s.z = s.pts.length ? z / s.pts.length : 0;
    }
    return all.sort((a, b) => a.z - b.z || a.depth - b.depth);
}

   nudge one control, rebuild, and see whether a single point moved.
   if nothing moves, that control is inert in this configuration.   */
function fingerprint(P) {
    const st = buildStrokes(P);
    let a = st.length * 7.3;
    for (let i = 0; i < st.length; i++) {
        const ps = st[i].pts;
        a += ps.length * 0.37;
        for (let j = 0; j < ps.length; j++) {
            const x = ps[j][0], y = ps[j][1], z = ps[j][2];
            a += Math.abs(x) * 1.1 + Math.abs(y) * 2.3 + Math.abs(z) * 3.7
                + (x * x + y * y + z * z) * 0.13 + Math.abs(x * y) * 0.29;
        }
    }
    return a;
}

function runAudit(P) {
    const base = fingerprint(P);
    const dead = [];
    const tol = Math.max(1e-7, Math.abs(base) * 1e-9);
    const moved = (Q) => Math.abs(fingerprint(Q) - base) > tol;
    for (const [k, label, lo, hi, vis] of GEOM) {
        if (!vis(P))
            continue;
        const step = (hi - lo) * 0.17;
        let v = P[k] + step;
        if (v > hi)
            v = P[k] - step;
        if (!moved(Object.assign({}, P, { [k]: v })))
            dead.push(label);
    }
    const chans = [["mixW", "W"], ["mixK", "K"], ["mixZ", "Z"], ["mixV", "V"]];
    for (const [ch, tag] of chans) {
        if (ch === "mixW" && P.mode === "filament")
            continue;
        if (ch === "mixK" && P.mode === "petal")
            continue;
        const live = P[ch].slots.filter((x) => x.t !== "off").length;
        P[ch].slots.forEach((sl, i) => {
            if (sl.t === "off")
                return;
            const bump = (mut) => {
                const slots = P[ch].slots.map((x) => ({ t: x.t, w: x.w, p: Object.assign({}, x.p) }));
                mut(slots[i]);
                return Object.assign({}, P, { [ch]: { op: P[ch].op, slots } });
            };
            const T = ch === "mixW" ? W_T : ch === "mixK" ? K_T : ch === "mixZ" ? Z_T : V_T;
            if (!weightless(T, sl.t, P[ch].op) &&
                !moved(bump((x) => { x.w = x.w > 0.7 ? x.w - 0.4 : x.w + 0.4; })))
                dead.push(tag + " slot " + (i + 1) + " · " + sl.t + " weight");
            for (const key of Object.keys(sl.p)) {
                if (!moved(bump((x) => { x.p[key] = x.p[key] + Math.max(0.15, Math.abs(x.p[key]) * 0.4); })))
                    dead.push(tag + " · " + sl.t + " · " + key);
            }
        });
    }
    return dead;
}
