  let hookIndex = 0;
  let effectQueue = [];

  function createElement(type, props, ...children) {
    const next = Object.assign({}, props || {});
    if (children.length === 1) next.children = children[0];
    else if (children.length > 1) next.children = children;
    return { type, props: next };
  }

  function depsChanged(a, b) {
    if (!a || !b || a.length !== b.length) return true;
    for (let i = 0; i < a.length; i++) if (!Object.is(a[i], b[i])) return true;
    return false;
  }

  function useState(initial) {
    const i = hookIndex++;
    let slot = rootState.hooks[i];
    if (!slot || slot.kind !== "state") {
      slot = { kind: "state", value: typeof initial === "function" ? initial() : initial, setter: null };
      slot.setter = (next) => {
        const value = typeof next === "function" ? next(slot.value) : next;
        if (!Object.is(value, slot.value)) {
          slot.value = value;
          scheduleRender();
        }
      };
      rootState.hooks[i] = slot;
    }
    return [slot.value, slot.setter];
  }

  function useRef(initial) {
    const i = hookIndex++;
    let slot = rootState.hooks[i];
    if (!slot || slot.kind !== "ref") {
      slot = { kind: "ref", value: { current: initial } };
      rootState.hooks[i] = slot;
    }
    return slot.value;
  }

  function useMemo(factory, deps) {
    const i = hookIndex++;
    let slot = rootState.hooks[i];
    if (!slot || slot.kind !== "memo" || depsChanged(slot.deps, deps)) {
      slot = { kind: "memo", deps, value: factory() };
      rootState.hooks[i] = slot;
    }
    return slot.value;
  }

  function useCallback(fn, deps) {
    return useMemo(() => fn, deps);
  }

  function useEffect(fn, deps) {
    const i = hookIndex++;
    const slot = rootState.hooks[i];
    if (!slot || slot.kind !== "effect" || depsChanged(slot.deps, deps)) {
      effectQueue.push({ i, fn, deps, previous: slot });
    }
  }

  function flatten(input, out) {
    if (input == null || input === false || input === true) return;
    if (Array.isArray(input)) {
      for (const item of input) flatten(item, out);
      return;
    }
    if (typeof input === "string" || typeof input === "number") {
      out.push({ type: "#text", text: String(input), props: {}, children: [] });
      return;
    }
    if (typeof input.type === "function") {
      flatten(input.type(input.props || {}), out);
      return;
    }
    if (input.type === Fragment) {
      flatten(input.props && input.props.children, out);
      return;
    }
    const props = input.props || {};
    const children = [];
    flatten(props.children, children);
    out.push({ type: input.type, props, children, key: props.key, dom: null });

    }
  }

    oldProps = oldProps || {};
    newProps = newProps || {};
    dom.__miniListeners = dom.__miniListeners || {};

        const ev = eventName(dom, k);
        if (!newProps[k] || newProps[k] !== oldProps[k]) {

        }

        }
      } else if (k === "className") {

      } else if (k === "htmlFor") {

      } else if (k === "value") {

      } else if (v === true) {

        dom.setAttribute(k, "");
      } else if (v === false || v == null) {
        if (k in dom && typeof dom[k] === "boolean") dom[k] = false;
        dom.removeAttribute(k);
      } else if (k in dom && !k.includes("-")) {
        try { dom[k] = v; } catch (_) { dom.setAttribute(k, String(v)); }
      } else {
        dom.setAttribute(k, String(v));
      }
    }

    if (oldProps.ref && oldProps.ref !== newProps.ref && typeof oldProps.ref === "object") oldProps.ref.current = null;
    if (newProps.ref && typeof newProps.ref === "object") newProps.ref.current = dom;
  }

    if (vnode.type === "#text") {
      vnode.dom = document.createTextNode(vnode.text);
      return vnode.dom;
    }
    const dom = document.createElement(vnode.type);
    vnode.dom = dom;
    updateProps(dom, {}, vnode.props);
    for (const child of vnode.children) dom.appendChild(createDom(child));
    return dom;
  }

  function patch(parent, oldVNode, newVNode) {
    if (!oldVNode && newVNode) {
      parent.appendChild(createDom(newVNode));
      return;
    }
    if (oldVNode && !newVNode) {
      if (oldVNode.props && oldVNode.props.ref && typeof oldVNode.props.ref === "object") oldVNode.props.ref.current = null;
      parent.removeChild(oldVNode.dom);
      return;

    if (oldVNode.type !== newVNode.type) {
      const nextDom = createDom(newVNode);
      if (oldVNode.props && oldVNode.props.ref && typeof oldVNode.props.ref === "object") oldVNode.props.ref.current = null;
      parent.replaceChild(nextDom, oldVNode.dom);
      return;
    }
    if (newVNode.type === "#text") {
      newVNode.dom = oldVNode.dom;
      if (oldVNode.text !== newVNode.text) oldVNode.dom.nodeValue = newVNode.text;
      return;
    }

    const dom = oldVNode.dom;
    newVNode.dom = dom;
    updateProps(dom, oldVNode.props, newVNode.props);

    const oldChildren = oldVNode.children;
    const newChildren = newVNode.children;
    const common = Math.min(oldChildren.length, newChildren.length);
    for (let i = 0; i < common; i++) patch(dom, oldChildren[i], newChildren[i]);
    for (let i = oldChildren.length - 1; i >= newChildren.length; i--) patch(dom, oldChildren[i], null);
    for (let i = oldChildren.length; i < newChildren.length; i++) patch(dom, null, newChildren[i]);
  }

  function flushEffects() {
    const queue = effectQueue;
    effectQueue = [];
    for (const item of queue) {
      if (item.previous && typeof item.previous.cleanup === "function") {
        try { item.previous.cleanup(); } catch (err) { console.error(err); }
      }
      let cleanup;
      try { cleanup = item.fn(); } catch (err) { console.error(err); }
      rootState.hooks[item.i] = { kind: "effect", deps: item.deps, cleanup };
    }
  }

  function performRender() {
    if (!rootState) return;
    hookIndex = 0;
    effectQueue = [];
    const expanded = [];
    flatten(rootState.rawVNode, expanded);
    const nextVNode = expanded[0] || null;
    patch(rootState.container, rootState.vnode, nextVNode);
    rootState.vnode = nextVNode;
    flushEffects();
  }

  function scheduleRender() {
    if (!rootState || rootState.scheduled) return;
    rootState.scheduled = true;
    queueMicrotask(() => {
      if (!rootState) return;
      rootState.scheduled = false;
      performRender();
    });
  }

  const React = { createElement, Fragment, useState, useRef, useEffect, useMemo, useCallback };
  const ReactDOM = {
    createRoot(container) {
      return {
        render(vnode) {
          rootState = { container, rawVNode: vnode, vnode: null, hooks: [], scheduled: false };
          performRender();
        }
      };
    }
  };

  global.React = React;
  global.ReactDOM = ReactDOM;
})(window);
